    <script>
        function filtrarCategoria(categoria) {
            const articulos = document.querySelectorAll('.articulo');
            const botones = document.querySelectorAll('.btn-categoria');

            // Cambiar estado activo de los botones
            botones.forEach(btn => btn.classList.remove('activo'));
            event.target.classList.add('activo');

            // Mostrar/Ocultar artículos
            articulos.forEach(art => {
                if (categoria === 'todas' || art.classList.contains(categoria)) {
                    art.classList.add('mostrar');
                } else {
                    art.classList.remove('mostrar');
                }
            });
        }

        // Mostrar todos los artículos por defecto al cargar la página
        filtrarCategoria('todas');
    </script>
